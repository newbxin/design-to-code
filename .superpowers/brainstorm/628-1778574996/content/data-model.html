Schema node contract
{
  id: "node-1",
  name: "banner",
  tag: "div" | "img" | "span" | "button",
  type: "container" | "text" | "image" | "shape" | "skeleton",
  bounds: { x, y, width, height },
  styles: { ...cssLikeProperties },
  text?: "按钮文案",
  asset?: {
    usage: "img" | "background",
    sourceImage: "page-capture-url-or-path",
    rect: { x, y, width, height }
  },
  children: []
}

Important rule

Style editing patches node.styles in the current frontend schema. Code preview and save both regenerate from that current schema. Nothing in this schema is persisted in MVP.
